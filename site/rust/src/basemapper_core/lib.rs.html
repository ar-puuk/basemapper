102/// The HTTP client for styles, tiles, glyphs and sprites.
103pub(crate) fn http_client(timeout: std::time::Duration) -> Result<reqwest::Client, BasemapError> {
104    reqwest::Client::builder()
105        .timeout(timeout)
106        .use_rustls_tls()
107        .user_agent(USER_AGENT)
108        .build()
109        .map_err(|e| BasemapError::RenderError(e.to_string()))

112/// Render a basemap tile mosaic to a flat RGBA byte array.
113///
114/// Supports anti-meridian crossing bboxes (xmin > xmax in EPSG:3857).
115/// Such bboxes are split internally into west and east halves, each rendered
116/// at proportional pixel width, then stitched horizontally.
117pub fn render(request: RenderRequest) -> Result<RenderResult, BasemapError> {
118    request.validate()?;
119    RUNTIME.block_on(render_async(request))
120}

122/// Render several styles over the same view and composite them bottom to
123/// top (straight-alpha source-over), e.g. a basemap published as separate
124/// services. All requests must share bbox, size and pixel ratio.
125pub fn render_stack(requests: Vec<RenderRequest>) -> Result<RenderResult, BasemapError> {
126    let first = requests.first().ok_or_else(|| BasemapError::RenderError("no styles to render".into()))?;
127    let (bbox, w, h, r) = (first.bbox, first.width, first.height, first.pixel_ratio);
128    if requests.iter().any(|q| q.bbox != bbox || q.width != w || q.height != h || q.pixel_ratio != r) {
129        return Err(BasemapError::RenderError("all styles in a stack must share bbox, size and pixel ratio".into()));
130    }
131    let mut acc: Option<RenderResult> = None;
132    for q in requests {
133        let top = render(q)?;
134        // A name warns only when it matches nothing in any style of the stack.
135        let top_warnings = top.warnings.clone();
136        let acc_warnings = acc.as_ref().map(|a| a.warnings.clone());
137        acc = Some(match acc {
138            None => top,
139            Some(mut below) => {
140                if below.pixels.len() != top.pixels.len() {
141                    return Err(BasemapError::RenderError("stacked renders differ in size".into()));
142                }
143                composite_over(&mut below.pixels, &top.pixels);
144                below.zoom_used = below.zoom_used.max(top.zoom_used);
145                below
146            }
147        });
148        if let (Some(a), Some(prev)) = (acc.as_mut(), acc_warnings) {
149            a.warnings = prev.into_iter().filter(|w| top_warnings.contains(w)).collect();
150        }
151    }
152    Ok(acc.expect("at least one request"))
153}

155/// `dst = src over dst`, both straight-alpha RGBA.
156fn composite_over(dst: &mut [u8], src: &[u8]) {
157    for (o, s) in dst.chunks_exact_mut(4).zip(src.chunks_exact(4)) {
158        let (sa, da) = (s[3] as f64 / 255.0, o[3] as f64 / 255.0);
159        let a = sa + da * (1.0 - sa);
160        if a <= 0.0 {
161            o.copy_from_slice(&[0, 0, 0, 0]);
162            continue;
163        }
164        for i in 0..3 {
165            let c = (s[i] as f64 / 255.0 * sa + o[i] as f64 / 255.0 * da * (1.0 - sa)) / a;
166            o[i] = (c * 255.0).round().clamp(0.0, 255.0) as u8;
167        }
168        o[3] = (a * 255.0).round() as u8;
169    }
170}

211/// A style's layers in drawing order; with a filter, the layers a render
212/// with that filter draws (none when it leaves nothing).
213///
214/// `style` is anything [`render`] accepts: a style, or tile data (a tile
215/// URL, TileJSON, PMTiles archive or ArcGIS service) with its built style.
216pub fn list_layers(style: &str, auth_token: Option<&str>, filter: &LayerFilter) -> Result<Vec<LayerInfo>, BasemapError> {
217    let json = RUNTIME.block_on(async {
218        let client = http_client(std::time::Duration::from_secs(30))?;
219        StyleInput::detect(style).resolve(&client, auth_token).await
220    })?;
221    if let Some(f) = &filter.features {
222        features::validate_features(f)?;
223    }
224    let (selection, _) = apply_filters(&json, filter.layers.as_deref(), filter.labels, filter.features.as_deref())?;
225    let v: serde_json::Value =
226        serde_json::from_str(&selection.style).map_err(|e| BasemapError::StyleParseError(e.to_string()))?;
227    Ok(v["layers"]
228        .as_array()
229        .map(|ls| {
230            ls.iter()
231                .map(|l| LayerInfo {
232                    id: l["id"].as_str().unwrap_or("").to_owned(),
233                    kind: l["type"].as_str().unwrap_or("").to_owned(),
234                    source_layer: l["source-layer"].as_str().map(str::to_owned),
235                    feature: features::layer_feature(l).to_owned(),
236                })
237                .collect()
238        })
239        .unwrap_or_default())
240}

253/// Encode straight-alpha RGBA pixels as a PNG file.
254pub fn encode_png(pixels: &[u8], width: u32, height: u32) -> Result<Vec<u8>, BasemapError> {
255    use image::ImageEncoder;
256    let mut buf = Vec::new();
257    image::codecs::png::PngEncoder::new(&mut buf)
258        .write_image(pixels, width, height, image::ExtendedColorType::Rgba8)
259        .map_err(|e| BasemapError::RenderError(format!("PNG encoding failed: {e}")))?;
260    Ok(buf)
261}

267/// The style with `layers`, `labels` and `features` applied, and warnings
268/// for the names that matched no layer of it.
269fn apply_filters(
270    raw_json: &str,
271    layers: Option<&[String]>,
272    labels: Labels,
273    features: Option<&[String]>,
274) -> Result<(features::Selection, Vec<String>), BasemapError> {
275    let (json, unmatched_layers) = match layers {
276        Some(l) => style::filter_layers(raw_json, l)?,
277        None => (raw_json.to_owned(), Vec::new()),
278    };
279    let selection = features::select(&json, labels, features.unwrap_or(&[]))?;
280    let mut warnings: Vec<String> =
281        unmatched_layers.iter().map(|l| format!("layers: {l:?} matches no layer")).collect();
282    warnings.extend(selection.unmatched.iter().map(|f| format!("features: {f:?} matches no layer")));
283    Ok((selection, warnings))
284}

286/// The style a request draws: its filters applied (an error when they leave
287/// nothing to draw), with warnings for names that matched no layer.
288fn select_style_layers(raw_json: &str, request: &RenderRequest) -> Result<(String, Vec<String>), BasemapError> {
289    let (selection, warnings) =
290        apply_filters(raw_json, request.layers.as_deref(), request.labels, request.features.as_deref())?;
291    // `layers` may leave nothing (as before); `labels` / `features` emptying a
292    // style that had layers is an error.
293    if selection.empty && !selection.present.is_empty() {
294        return Err(BasemapError::InvalidLayerFilter(format!(
295            "labels and features leave nothing of this style to draw; its features are {}",
296            selection.present.join(", ")
297        )));
298    }
299    Ok((selection.style, warnings))
300}

302async fn render_async(request: RenderRequest) -> Result<RenderResult, BasemapError> {
303    let client = http_client(std::time::Duration::from_millis(request.tile_timeout_ms as u64))?;
304
305    // Resolve style once — shared across both halves for anti-meridian renders.
306    let raw_json = request
307        .style_input
308        .resolve(&client, request.auth_token.as_deref())
309        .await?;
310    style::validate_maplibre_style(&raw_json)?;
311
312    let (style_json, warnings) = select_style_layers(&raw_json, &request)?;
313    let mut result = render_style(request, client, style_json).await?;
314    result.warnings = warnings;
315    Ok(result)
316}

318/// Render a request with its resolved, filtered style.
319async fn render_style(request: RenderRequest, client: reqwest::Client, style_json: String) -> Result<RenderResult, BasemapError> {
320    // Extract tile source (may fetch TileJSON).
321    let style_val: serde_json::Value = serde_json::from_str(&style_json)
322        .map_err(|e| BasemapError::StyleParseError(e.to_string()))?;
323    // Filtered styles and label-only tiles go over other maps: no background fill.
324    let transparent = request.layers.is_some()
325        || request.labels != Labels::All
326        || request.features.is_some()
327        || style_val.pointer("/metadata/basemapper:transparent") == Some(&serde_json::Value::Bool(true));
328    let style_base_url = match &request.style_input {
329        StyleInput::Url(u) => Some(u.as_str()),
330        StyleInput::InlineJson(_) => None,
331    };
332    let tile_source = extract_tile_source_from_style(
333        &style_val,
334        &client,
335        request.auth_token.as_deref(),
336        style_base_url,
337    )
338    .await?;
339
340    // Compute zoom for the full bbox — handles anti-meridian via compute_zoom.
341    let zoom = {
342        let raw = request.tile_zoom.unwrap_or_else(|| {
343            let mut z = bbox::raster_tile_zoom(request.bbox, request.width, raster_tile_size(&style_val));
344            while z > 0 && bbox::tile_count(request.bbox, z) > request.max_tiles {
345                z -= 1;
346            }
347            z
348        });
349        tile_source.maxzoom().map_or(raw, |mz| raw.min(mz))
350    };
351
352    // Guard: tile_count handles anti-meridian correctly now.
353    let total_tiles = bbox::tile_count(request.bbox, zoom);
354    if total_tiles > request.max_tiles {
355        return Err(BasemapError::MaxTilesExceeded {
356            requested: total_tiles,
357            limit: request.max_tiles,
358        });
359    }
360
361    let [xmin, _ymin, xmax, _ymax] = request.bbox;
362    let half = 20_037_508.342_789_244;
363    let wraps = xmin < xmax && (xmin < -half || xmax > half);
364    let is_vector = matches!(tile_source, TileSource::MapboxVectorTile { .. });
365    if xmin > xmax {
366        render_antimeridian_split(&request, &client, &style_json, &tile_source, transparent, zoom)
367            .await
368    } else if wraps && !is_vector {
369        // Raster sources: express a view past ±180° as an anti-meridian bbox.
370        // (Vector sources draw the wrapped world copy directly.)
371        let mut req = request.clone();
372        if req.bbox[0] < -half {
373            req.bbox[0] += 2.0 * half;
374        }
375        if req.bbox[2] > half {
376            req.bbox[2] -= 2.0 * half;
377        }
378        render_antimeridian_split(&req, &client, &style_json, &tile_source, transparent, zoom).await
379    } else {
380        render_single_bbox(
381            &request,
382            &client,
383            &style_json,
384            &tile_source,
385            transparent,
386            request.bbox,
387            request.width,
388            request.height,
389            zoom,
390        )
391        .await
392    }
393}

395/// Render one non-crossing bbox at the given zoom level and pixel dimensions.
396///
397/// Called both for normal bboxes and for each half of an anti-meridian split.
398async fn render_single_bbox(
399    request: &RenderRequest,
400    client: &reqwest::Client,
401    style_json: &str,
402    tile_source: &TileSource,
403    transparent: bool,
404    bbox: [f64; 4],
405    width: u32,
406    height: u32,
407    zoom: u8,
408) -> Result<RenderResult, BasemapError> {
409    let bounds = SpatialBounds {
410        xmin: bbox[0],
411        ymin: bbox[1],
412        xmax: bbox[2],
413        ymax: bbox[3],
414        crs_epsg: 3857,
415    };
416
417    // Vector tiles: in-house CPU renderer (core/src/vector/).
418    if matches!(tile_source, TileSource::MapboxVectorTile { .. }) {
419        // MapLibre convention: style evaluated at the true scale zoom; vector
420        // tiles (512 px) fetched at floor(scale zoom), clamped to the source
421        // maxzoom. An explicit request zoom is on the 256 px raster grid, which
422        // is one level above the equivalent 512 px tile zoom.
423        let (style_val, placed, assets, tz) =
424            fetch_vector_inputs(request, client, style_json, tile_source, bbox, width).await?;
425        let out = vector::render::render(&style_val, &placed, &assets, bbox, width, height, request.pixel_ratio)
426            .map_err(BasemapError::RenderError)?;
427        if !out.skipped_layer_types.is_empty() {
428            log::warn!(
429                "vector renderer does not draw these layer types yet: {}",
430                out.skipped_layer_types.join(", ")
431            );
432        }
433        return Ok(RenderResult {
434            bounds,
435            zoom_used: tz.saturating_add(1),
436            warnings: Vec::new(),
437            width: out.width,
438            height: out.height,
439            pixels: out.pixels,
440        });
441    }
442    // Raster sources: tiles chosen for the CSS size, composited at device size.
443    let (width, height) = (
444        vector::render::device_px(width, request.pixel_ratio),
445        vector::render::device_px(height, request.pixel_ratio),
446    );
447    // PMTiles: fetch via HTTP range requests.
448    if let TileSource::PMTiles { ref url, .. } = tile_source {
449        let coords = tile_fetcher::build_tile_coords(bbox, zoom);
450        let tiles = fetch_pmtiles_tiles(url, &coords, client).await?;
451        let canvas =
452            renderer::composite_raster_tiles(&tiles, bbox, zoom, width, height, transparent, request.pixel_ratio)?;
453        return Ok(RenderResult {
454            bounds,
455            zoom_used: zoom,
456            warnings: Vec::new(),
457            width,
458            height,
459            pixels: canvas.into_raw(),
460        });
461    }
462
463    // Raster XYZ (also WMS, through {bbox-epsg-3857}).
464    let coords = tile_fetcher::build_tile_coords(bbox, zoom);
465    let urls = tile_fetcher::build_tile_urls(tile_source, &coords, request.pixel_ratio);
466    let tiles = tile_fetcher::fetch_all_tiles(
467        client,
468        urls,
469        tile_source,
470        request.tile_timeout_ms,
471        request.tile_concurrency,
472        request.fail_on_tile_error,
473    )
474    .await?;
475    let canvas =
476        renderer::composite_raster_tiles(&tiles, bbox, zoom, width, height, transparent, request.pixel_ratio)?;
477    Ok(RenderResult {
478        bounds,
479        zoom_used: zoom,
480        warnings: Vec::new(),
481        width,
482        height,
483        pixels: canvas.into_raw(),
484    })
485}

487/// Split an anti-meridian bbox into west and east halves, render each, then
488/// stitch the pixel arrays horizontally (west first).
489async fn render_antimeridian_split(
490    request: &RenderRequest,
491    client: &reqwest::Client,
492    style_json: &str,
493    tile_source: &TileSource,
494    transparent: bool,
495    zoom: u8,
496) -> Result<RenderResult, BasemapError> {
497    const EARTH_HALF: f64 = 20_037_508.342_789_244;
498    let [xmin, ymin, xmax, ymax] = request.bbox;
499
500    // West half: from xmin → +EARTH_HALF (right edge of the tile grid).
501    // East half: from −EARTH_HALF (left edge) → xmax.
502    let west_bbox = [xmin, ymin, EARTH_HALF, ymax];
503    let east_bbox = [-EARTH_HALF, ymin, xmax, ymax];
504
505    let west_m = EARTH_HALF - xmin;
506    let east_m = xmax + EARTH_HALF;
507    let total_m = west_m + east_m;
508
509    // Proportional pixel widths; guarantee at least 1 px per half.
510    let west_px = ((request.width as f64 * west_m / total_m).round() as u32).max(1);
511    let east_px = request.width.saturating_sub(west_px).max(1);
512
513    let west = render_single_bbox(
514        request, client, style_json, tile_source, transparent,
515        west_bbox, west_px, request.height, zoom,
516    )
517    .await?;
518    let east = render_single_bbox(
519        request, client, style_json, tile_source, transparent,
520        east_bbox, east_px, request.height, zoom,
521    )
522    .await?;
523
524    // Stitch row by row: west pixels then east pixels per scanline.
525    let row_w = west.width as usize * 4;
526    let row_e = east.width as usize * 4;
527    let row_out = row_w + row_e;
528    let rows = west.height.min(east.height) as usize;
529    let mut pixels = vec![0u8; row_out * rows];
530    for row in 0..rows {
531        let out = &mut pixels[row * row_out..][..row_out];
532        out[..row_w].copy_from_slice(&west.pixels[row * row_w..][..row_w]);
533        out[row_w..].copy_from_slice(&east.pixels[row * row_e..][..row_e]);
534    }
535
536    Ok(RenderResult {
537        bounds: SpatialBounds { xmin, ymin, xmax, ymax, crs_epsg: 3857 },
538        zoom_used: zoom,
539        warnings: Vec::new(),
540        width: west.width + east.width,
541        height: rows as u32,
542        pixels,
543    })
544}

558    auth_token: Option<&str>,
559    style_base_url: Option<&str>,
560) -> Result<TileSource, BasemapError> {
561    let sources = style
562        .get("sources")
563        .and_then(|s| s.as_object())
564        .ok_or_else(|| BasemapError::StyleParseError("missing \"sources\" object".into()))?;
565    static GEOJSON_ONLY: Lazy<serde_json::Value> = Lazy::new(|| serde_json::json!({"type": "vector", "tiles": [""], "maxzoom": 22}));
566
567    // Prefer vector sources over raster; prevents multi-source styles from
568    // routing through the raster pipeline when a vector source is present.
569    let vector_entry = sources.values().find_map(|s| {
570        if s.get("type").and_then(|t| t.as_str()) == Some("vector") {
571            Some(("vector", s))
572        } else {
573            None
574        }
575    });
576    // Sources only the vector path draws (GeoJSON, DEM for hillshade): such
577    // styles render through it even without a vector source.
578    let needs_vector_path = sources.values().any(|s| matches!(s["type"].as_str(), Some("geojson" | "raster-dem")));
579    let (src_type, source) = if let Some(entry) = vector_entry {
580        entry
581    } else if needs_vector_path {
582        ("vector", &*GEOJSON_ONLY)
583    } else {
584        sources
585            .values()
586            .find_map(|s| {
587                let t = s.get("type").and_then(|t| t.as_str()).unwrap_or("raster");
588                if matches!(t, "geojson" | "image" | "video") {
589                    None
590                } else {
591                    Some((t, s))
592                }
593            })
594            .or_else(|| {
595                // GeoJSON-only (or source-less) styles render through the
596                // vector path, which reads every source the layers use.
597                (sources.is_empty() || sources.values().any(|s| s["type"] == "geojson"))
598                    .then(|| ("vector", &*GEOJSON_ONLY))
599            })
600            .ok_or_else(|| {
601                BasemapError::StyleParseError("no raster or vector tile source found".into())
602            })?
603    };
604
605    let resolved = resolve_source(source, src_type, client, auth_token, style_base_url).await?.source;
606    Ok(match resolved {
607        // A vector archive: the vector path, which fetches every source
608        // itself (from the archive), with the archive's max zoom.
609        TileSource::PMTiles { maxzoom, .. } if src_type == "vector" => {
610            TileSource::MapboxVectorTile { urls: TileUrls::default(), api_key: None, maxzoom }
611        }
612        other => other,
613    })
614}

616/// A style source resolved to its tile URL template (inline `tiles` or
617/// TileJSON), with the zoom range and tile size MapLibre uses for it.
618struct ResolvedSource {
619    source: TileSource,
620    minzoom: u8,
621    /// Raster tile size in px (MapLibre's default for raster sources: 512).
622    tile_size: f64,
623}
624
625/// The URL a TileJSON's relative `tiles` resolve against. An ArcGIS
626/// service's JSON (`.../VectorTileServer`) lists them relative to the
627/// service itself (`tile/{z}/{y}/{x}.pbf`), as if its URL ended in `/`.
628fn tilejson_base(url: &str) -> String {
629    let (path, query) = url.split_once('?').map_or((url, None), |(p, q)| (p, Some(q)));
630    let lower = path.to_ascii_lowercase();
631    if lower.ends_with("/vectortileserver") || lower.ends_with("/mapserver") {
632        match query {
633            Some(q) => format!("{path}/?{q}"),
634            None => format!("{path}/"),
635        }
636    } else {
637        url.to_owned()
638    }
639}
640
641async fn resolve_source(
642    source: &serde_json::Value,
643    src_type: &str,
644    client: &reqwest::Client,
645    auth_token: Option<&str>,
646    style_base_url: Option<&str>,
647) -> Result<ResolvedSource, BasemapError> {
648    let num = |v: &serde_json::Value, k: &str| v.get(k).and_then(|x| x.as_f64());
649    let tile_list = |v: &serde_json::Value| -> Vec<String> {
650        v.get("tiles")
651            .and_then(|t| t.as_array())
652            .map(|a| a.iter().filter_map(|u| u.as_str().map(str::to_owned)).collect())
653            .unwrap_or_default()
654    };
655    let mut minzoom = num(source, "minzoom");
656    let mut tile_size = num(source, "tileSize");
657    // `scheme` from the source, else from its TileJSON (as MapLibre's
658    // loadTileJson, where explicit source options take precedence).
659    let mut scheme = source.get("scheme").and_then(|v| v.as_str()).map(str::to_owned);
660    // Resolve the tile URL templates and maxzoom.
661    let (tiles, source_maxzoom): (Vec<String>, Option<u8>) = if !tile_list(source).is_empty() {
662        let mz = source
663            .get("maxzoom")
664            .and_then(|v| v.as_u64())
665            .map(|v| v.min(22) as u8);
666        (tile_list(source), mz)
667    } else if let Some(archive) = source.get("url").and_then(|u| u.as_str()).and_then(|u| u.strip_prefix("pmtiles://")) {
668        // The pmtiles protocol's TileJSON: zoom range from the archive
669        // header; the source's own minzoom / maxzoom take precedence.
670        let url = resolve_url(archive, style_base_url);
671        let reader = open_pmtiles(&url, client).await?;
672        let header = reader.get_header();
673        minzoom = minzoom.or(Some(header.min_zoom as f64));
674        let mz = source
675            .get("maxzoom")
676            .and_then(|v| v.as_u64())
677            .map_or(header.max_zoom, |v| v as u8)
678            .min(22);
679        (vec![format!("pmtiles://{url}")], Some(mz))
680    } else if let Some(tilejson_url_raw) = source.get("url").and_then(|u| u.as_str()) {
681        let tilejson_url = resolve_url(tilejson_url_raw, style_base_url);
682        let mut tilejson_req = client.get(&tilejson_url);
683        if let Some(token) = auth_token {
684            tilejson_req = tilejson_req.header("Authorization", format!("Bearer {token}"));
685        }
686        let resp = tilejson_req
687            .send()
688            .await
689            .map_err(|_| BasemapError::StyleFetchFailed {
690                url: tilejson_url.clone(),
691                status: 0,
692            })?;
693        if !resp.status().is_success() {
694            return Err(BasemapError::StyleFetchFailed {
695                url: tilejson_url.clone(),
696                status: resp.status().as_u16(),
697            });
698        }
699        let tilejson: serde_json::Value = resp.json().await.map_err(|_| {
700            BasemapError::StyleParseError("could not parse TileJSON response".into())
701        })?;
702        let base = tilejson_base(&tilejson_url);
703        let tiles: Vec<String> = tile_list(&tilejson).iter().map(|t| resolve_url(t, Some(&base))).collect();
704        if tiles.is_empty() {
705            return Err(BasemapError::StyleParseError("TileJSON missing \"tiles\" array".into()));
706        }
707        let mz = source
708            .get("maxzoom")
709            .or_else(|| tilejson.get("maxzoom"))
710            .and_then(|v| v.as_u64())
711            .map(|v| v.min(22) as u8);
712        minzoom = minzoom.or_else(|| num(&tilejson, "minzoom"));
713        tile_size = tile_size.or_else(|| num(&tilejson, "tileSize"));
714        scheme = scheme.or_else(|| tilejson.get("scheme").and_then(|v| v.as_str()).map(str::to_owned));
715        (tiles, mz)
716    } else {
717        return Err(BasemapError::StyleParseError(
718            "source has neither \"tiles\" array nor \"url\" TileJSON reference".into(),
719        ));
720    };
721
722    let urls = TileUrls { templates: tiles, tms: scheme.as_deref() == Some("tms") };
723    let source = if let Some(archive) = urls.templates.first().and_then(|t| t.strip_prefix("pmtiles://")) {
724        TileSource::PMTiles { url: archive.to_owned(), maxzoom: source_maxzoom }
725    } else if src_type == "vector" {
726        TileSource::MapboxVectorTile {
727            urls,
728            api_key: auth_token.map(str::to_owned),
729            maxzoom: source_maxzoom,
730        }
731    } else {
732        TileSource::XyzRaster {
733            urls,
734            auth_header: auth_token.map(|t| format!("Bearer {t}")),
735            maxzoom: source_maxzoom,
736        }
737    };
738    Ok(ResolvedSource {
739        source,
740        minzoom: minzoom.unwrap_or(0.0).clamp(0.0, 22.0) as u8,
741        tile_size: tile_size.unwrap_or(512.0),
742    })
743}

745/// Fetch tiles from a PMTiles archive over HTTP using range requests.
746///
747/// Opens the archive (header + root directory via two range requests), then
748/// fetches each requested tile.  Tiles absent from the archive (sparse
749/// coverage) are silently skipped.
750async fn fetch_pmtiles_tiles(
751    url: &str,
752    coords: &[tile_fetcher::TileCoord],
753    client: &reqwest::Client,
754) -> Result<Vec<(tile_fetcher::TileCoord, tile_fetcher::TileData)>, BasemapError> {
755    let reader = std::sync::Arc::new(open_pmtiles(url, client).await?);
756    // Tiles are fetched concurrently and decompressed (tile_compression).
757    let mut set = tokio::task::JoinSet::new();
758    for (i, &coord) in coords.iter().enumerate() {
759        let pm_coord = pmtiles::TileCoord::new(coord.z, coord.x, coord.y)
760            .map_err(|e| BasemapError::RenderError(format!("PMTiles coord error: {e}")))?;
761        let reader = reader.clone();
762        set.spawn(async move { (i, coord, reader.get_tile_decompressed(pm_coord).await) });
763    }
764    let mut results = Vec::new();
765    while let Some(joined) = set.join_next().await {
766        let Ok((i, coord, res)) = joined else { continue };
767        match res {
768            Ok(Some(bytes)) => results.push((i, coord, bytes.to_vec())),
769            Ok(None) => {}
770            Err(e) => {
771                log::warn!(
772                    "PMTiles tile {}/{}/{} fetch error (skipped): {e}",
773                    coord.z,
774                    coord.x,
775                    coord.y,
776                );
777            }
778        }
779    }
780    results.sort_by_key(|r| r.0);
781    Ok(results.into_iter().map(|(_, c, d)| (c, d)).collect())
782}

784/// Open a PMTiles archive over HTTP (header and root directory).
785pub(crate) async fn open_pmtiles(
786    url: &str,
787    client: &reqwest::Client,
788) -> Result<pmtiles::AsyncPmTilesReader<tile_fetcher::ArchiveBackend>, BasemapError> {
789    let backend = tile_fetcher::ArchiveBackend::new(url.to_owned(), client.clone());
790    pmtiles::AsyncPmTilesReader::try_from_source(backend)
791        .await
792        .map_err(|e| BasemapError::RenderError(format!("PMTiles open failed ({url}): {e}")))
793}

795/// Resolve a potentially-relative URL against a base URL.
796///
797/// `reqwest::Url::join` percent-encodes `{` and `}` in tile template
798/// placeholders, so we decode them back after joining.
799pub(crate) fn resolve_url(url: &str, base: Option<&str>) -> String {
800    if url.starts_with("http://") || url.starts_with("https://") {
801        return url.to_owned();
802    }
803    if let Some(base_str) = base {
804        if let Ok(base_parsed) = reqwest::Url::parse(base_str) {
805            if let Ok(resolved) = base_parsed.join(url) {
806                return resolved
807                    .to_string()
808                    .replace("%7B", "{")
809                    .replace("%7D", "}")
810                    .replace("%7b", "{")
811                    .replace("%7d", "}");
812            }
813        }
814    }
815    url.to_owned()
816}

834/// Fetch the glyph PBF ranges the visible text labels need, from the style's
835/// `glyphs` URL template. Failures are logged and skipped (labels whose glyphs
836/// are missing are not drawn), as MapLibre does.
837async fn fetch_glyphs(
838    client: &reqwest::Client,
839    style: &serde_json::Value,
840    style_base_url: Option<&str>,
841    tiles: &[(tile_fetcher::TileCoord, tile_fetcher::TileData)],
842    zoom: f64,
843) -> vector::symbol::GlyphMap {
844    let mut map = vector::symbol::GlyphMap::default();
845    let Some(template) = style["glyphs"].as_str() else { return map };
846    let template = resolve_url(template, style_base_url);
847    let decoded: Vec<vector::mvt::Tile> = tiles.iter().filter_map(|(_, b)| vector::mvt::decode(b).ok()).collect();
848    let mut needed: Vec<(String, u32)> =
849        vector::symbol::required_glyph_ranges(style, decoded.iter(), zoom).into_iter().collect();
850    needed.sort();
851    let mut set = tokio::task::JoinSet::new();
852    for (stack, start) in needed {
853        let url = template
854            .replace("{fontstack}", &stack)
855            .replace("{range}", &format!("{}-{}", start, start + 255));
856        let client = client.clone();
857        set.spawn(async move {
858            let resp = client.get(&url).send().await.ok()?;
859            if !resp.status().is_success() {
860                log::warn!("glyphs {url}: HTTP {}", resp.status());
861                return None;
862            }
863            Some((stack, resp.bytes().await.ok()?.to_vec()))
864        });
865    }
866    let mut results = Vec::new();
867    while let Some(r) = set.join_next().await {
868        if let Ok(Some(x)) = r {
869            results.push(x);
870        }
871    }
872    results.sort_by(|a, b| a.0.cmp(&b.0));
873    for (stack, bytes) in results {
874        if let Err(e) = map.add_pbf(&stack, &bytes) {
875            log::warn!("glyphs for {stack}: {e}");
876        }
877    }
878    map
879}

881/// Fetch the style's 1x sprite (`<sprite>.json` + `<sprite>.png`). A missing
882/// or broken sprite only disables icons.
883/// The style's sprite; above pixel ratio 1 the `@2x` sheet (as MapLibre
884/// loads it), falling back to 1x. Multi-sprite styles (`[{id, url}, ...]`)
885/// load every sheet; images of sprites other than `default` are named
886/// `id:name`, as MapLibre addresses them.
887async fn fetch_sprite(
888    client: &reqwest::Client,
889    style: &serde_json::Value,
890    style_base_url: Option<&str>,
891    pixel_ratio: f64,
892) -> vector::symbol::Sprite {
893    let entries: Vec<(String, String)> = match &style["sprite"] {
894        serde_json::Value::String(u) => vec![("default".to_owned(), u.clone())],
895        serde_json::Value::Array(list) => list
896            .iter()
897            .filter_map(|e| Some((e["id"].as_str()?.to_owned(), e["url"].as_str()?.to_owned())))
898            .collect(),
899        _ => Vec::new(),
900    };
901    let mut sprite = vector::symbol::Sprite::default();
902    for (id, url) in entries {
903        let base = resolve_url(&url, style_base_url);
904        let get = |url: String| {
905            let client = client.clone();
906            async move {
907                let resp = client.get(&url).send().await.ok()?;
908                if !resp.status().is_success() {
909                    log::warn!("sprite {url}: HTTP {}", resp.status());
910                    return None;
911                }
912                resp.bytes().await.ok().map(|b| b.to_vec())
913            }
914        };
915        let mut fetched = None;
916        if pixel_ratio > 1.0 {
917            let (json, png) = tokio::join!(get(format!("{base}@2x.json")), get(format!("{base}@2x.png")));
918            if let (Some(j), Some(p)) = (json, png) {
919                fetched = Some((j, p));
920            }
921        }
922        if fetched.is_none() {
923            let (json, png) = tokio::join!(get(format!("{base}.json")), get(format!("{base}.png")));
924            if let (Some(j), Some(p)) = (json, png) {
925                fetched = Some((j, p));
926            }
927        }
928        let Some((j, p)) = fetched else { continue };
929        match vector::symbol::Sprite::from_json_png(&j, &p) {
930            Ok(sheet) => {
931                for (name, img) in sheet.images {
932                    let key = if id == "default" { name } else { format!("{id}:{name}") };
933                    sprite.images.insert(key, img);
934                }
935            }
936            Err(e) => log::warn!("{e}"),
937        }
938    }
939    sprite.device_pixel_ratio = pixel_ratio;
940    sprite
941}

943/// Tiles for every tile source the style's layers use, plus glyphs and
944/// sprite, fetched as MapLibre would: the style is evaluated at the true
945/// scale zoom; vector sources (512 px tiles) at floor(zoom), raster sources
946/// at round(zoom + log2(512 / tileSize)), each clamped to its zoom range.
947/// `primary` is the style's main vector source: its tile zoom is returned.
948/// An explicit request zoom is on the 256 px raster grid, one level above
949/// the equivalent 512 px vector tile zoom.
950#[allow(clippy::type_complexity)]
951async fn fetch_vector_inputs(
952    request: &RenderRequest,
953    client: &reqwest::Client,
954    style_json: &str,
955    primary: &TileSource,
956    bbox: [f64; 4],
957    width: u32,
958) -> Result<(serde_json::Value, vector::render::SourceTiles, vector::symbol::SymbolAssets, u8), BasemapError> {
959    let style_val: serde_json::Value =
960        serde_json::from_str(style_json).map_err(|e| BasemapError::StyleParseError(e.to_string()))?;
961    // 3D terrain is out of scope for flat basemaps (with a straight-down
962    // camera MapLibre's terrain only resamples and displaces the map).
963    if style_val.get("terrain").is_some_and(|t| !t.is_null()) {
964        log::warn!("style terrain is not supported; the map is drawn flat");
965    }
966    let sz = vector::render::scale_zoom(bbox, width);
967    let vector_tz = |maxzoom: Option<u8>, minzoom: u8| -> Option<u8> {
968        let tz = match request.tile_zoom {
969            Some(z) => z.saturating_sub(1),
970            None => vector::render::tile_zoom(sz, None),
971        };
972        let tz = maxzoom.map_or(tz, |m| tz.min(m));
973        (tz >= minzoom).then_some(tz)
974    };
975    let primary_tz = vector_tz(primary.maxzoom(), 0).unwrap_or(0);
976
977    // Sources used by layers visible at this zoom, in style order.
978    let mut used: Vec<String> = Vec::new();
979    for l in style_val["layers"].as_array().map(|v| v.as_slice()).unwrap_or(&[]) {
980        if !vector::symbol::layer_visible(l, sz) {
981            continue;
982        }
983        if let Some(src) = l["source"].as_str() {
984            if !used.iter().any(|u| u == src) {
985                used.push(src.to_owned());
986            }
987        }
988    }
989    let mut sources = vector::render::SourceTiles::default();
990    let mut all_vector: Vec<(tile_fetcher::TileCoord, tile_fetcher::TileData)> = Vec::new();
991    for id in used {
992        let src = &style_val["sources"][&id];
993        let kind = src["type"].as_str().unwrap_or("");
994        if kind == "geojson" {
995            if src["cluster"].as_bool() == Some(true) {
996                log::warn!("source {id}: GeoJSON clustering is not supported; drawing unclustered features");
997            }
998            let data = match &src["data"] {
999                serde_json::Value::String(url) => {
1000                    let url = resolve_url(url, style_base_url_of(request));
1001                    match client.get(&url).send().await {
1002                        Ok(r) if r.status().is_success() => r.json::<serde_json::Value>().await.unwrap_or_default(),
1003                        Ok(r) => {
1004                            log::warn!("source {id}: GeoJSON {url}: HTTP {}", r.status());
1005                            continue;
1006                        }
1007                        Err(e) => {
1008                            log::warn!("source {id}: GeoJSON {url}: {e}");
1009                            continue;
1010                        }
1011                    }
1012                }
1013                other => other.clone(),
1014            };
1015            let opts = vector::geojson::GeoJsonOptions::from_source(src);
1016            let index = vector::geojson::GeoJsonIndex::new(&data, opts);
1017            let tz = match vector_tz(Some(opts.max_zoom), 0) {
1018                Some(z) => z,
1019                None => continue,
1020            };
1021            let tiles: Vec<_> = vector::render::covering_tiles(bbox, tz)
1022                .into_iter()
1023                .map(|(c, wrap)| (c, wrap, index.tile(c.z, c.x, c.y)))
1024                .collect();
1025            all_vector.extend(tiles.iter().map(|(c, _, b)| (*c, b.clone())));
1026            sources.vector.insert(id, tiles);
1027            continue;
1028        }
1029        if kind != "vector" && kind != "raster" && kind != "raster-dem" {
1030            if !kind.is_empty() {
1031                log::warn!("source {id}: type {kind} is not supported yet");
1032            }
1033            continue;
1034        }
1035        let resolved = resolve_source(src, kind, client, request.auth_token.as_deref(), style_base_url_of(request)).await?;
1036        let source = &resolved.source;
1037        let fetch = |coords: Vec<tile_fetcher::TileCoord>| async move {
1038            if let TileSource::PMTiles { url, .. } = source {
1039                return fetch_pmtiles_tiles(url, &coords, client).await;
1040            }
1041            let urls = tile_fetcher::build_tile_urls(source, &coords, request.pixel_ratio);
1042            tile_fetcher::fetch_all_tiles(
1043                client,
1044                urls,
1045                source,
1046                request.tile_timeout_ms,
1047                request.tile_concurrency,
1048                request.fail_on_tile_error,
1049            )
1050            .await
1051        };
1052        if kind == "vector" {
1053            let Some(tz) = vector_tz(resolved.source.maxzoom(), resolved.minzoom) else { continue };
1054            let covering = vector::render::covering_tiles(bbox, tz);
1055            let mut coords: Vec<tile_fetcher::TileCoord> = covering.iter().map(|c| c.0).collect();
1056            coords.sort_by_key(|c| (c.z, c.x, c.y));
1057            coords.dedup();
1058            let tiles = fetch(coords).await?;
1059            let by_coord: std::collections::HashMap<(u8, u32, u32), &tile_fetcher::TileData> =
1060                tiles.iter().map(|(c, d)| ((c.z, c.x, c.y), d)).collect();
1061            let placed: Vec<_> = covering
1062                .iter()
1063                .filter_map(|(c, wrap)| by_coord.get(&(c.z, c.x, c.y)).map(|d| (*c, *wrap, (*d).clone())))
1064                .collect();
1065            all_vector.extend(tiles);
1066            sources.vector.insert(id, placed);
1067        } else {
1068            let z = bbox::raster_tile_zoom(bbox, width, resolved.tile_size);
1069            let z = resolved.source.maxzoom().map_or(z, |m| z.min(m));
1070            if z < resolved.minzoom {
1071                continue;
1072            }
1073            let tiles = fetch(tile_fetcher::build_tile_coords(bbox, z)).await?;
1074            if kind == "raster-dem" {
1075                let unpack = vector::hillshade::unpack_vector(src);
1076                sources.dem.insert(id, vector::hillshade::DemTiles { zoom: z, tiles, unpack });
1077            } else {
1078                sources.raster.insert(id, vector::render::RasterTiles { zoom: z, tiles });
1079            }
1080        }
1081    }
1082    let assets = vector::symbol::SymbolAssets {
1083        glyphs: fetch_glyphs(client, &style_val, style_base_url_of(request), &all_vector, sz).await,
1084        sprite: fetch_sprite(client, &style_val, style_base_url_of(request), request.pixel_ratio).await,
1085    };
1086    Ok((style_val, sources, assets, primary_tz))
1087}

1103/// Render several styles over the same view as one SVG document, bottom to
1104/// top, each style in its own group (e.g. a basemap built from separate
1105/// services). Fills, lines and circles are vector paths; labels are glyph
1106/// outlines traced from the style's SDF glyphs; non-SDF icons are embedded
1107/// sprite images. All requests must share bbox, width and height, and use
1108/// vector tile sources.
1109pub fn render_svg_stack(requests: Vec<RenderRequest>) -> Result<String, BasemapError> {
1110    render_svg_stack_with(requests, &SvgOptions::default())
1111}
1112
1113/// [`render_svg_stack`] with options: label text as editable `<text>` with
1114/// embedded fonts (default) or as traced glyph outlines.
1115pub fn render_svg_stack_with(requests: Vec<RenderRequest>, options: &SvgOptions) -> Result<String, BasemapError> {
1116    use std::fmt::Write as _;
1117    let first = requests.first().ok_or_else(|| BasemapError::RenderError("no styles to render".into()))?;
1118    let (bbox, width, height) = (first.bbox, first.width, first.height);
1119    if bbox[0] > bbox[2] {
1120        return Err(BasemapError::InvalidBbox("SVG output does not support anti-meridian (xmin > xmax) bboxes".into()));
1121    }
1122    for request in &requests {
1123        if request.bbox != bbox || request.width != width || request.height != height {
1124            return Err(BasemapError::RenderError("all styles in an SVG stack must share bbox and size".into()));
1125        }
1126        request.validate()?;
1127    }
1128    let timeout = requests.iter().map(|r| r.tile_timeout_ms).max().unwrap_or(10_000);
1129    RUNTIME.block_on(async {
1130        let client = http_client(std::time::Duration::from_millis(timeout.max(15_000) as u64))?;
1131        // Every style's tiles and assets first, so the fonts of the whole
1132        // document are resolved (and embedded) once.
1133        let mut inputs = Vec::new();
1134        for request in &requests {
1135            let raw_json = request.style_input.resolve(&client, request.auth_token.as_deref()).await?;
1136            style::validate_maplibre_style(&raw_json)?;
1137            let (style_json, _) = select_style_layers(&raw_json, request)?;
1138            let style_val: serde_json::Value =
1139                serde_json::from_str(&style_json).map_err(|e| BasemapError::StyleParseError(e.to_string()))?;
1140            let tile_source =
1141                extract_tile_source_from_style(&style_val, &client, request.auth_token.as_deref(), style_base_url_of(request)).await?;
1142            // Raster-only styles take the same path: raster layers are
1143            // embedded as images (vector layers are paths and text).
1144            let (style_val, sources, assets, _) =
1145                fetch_vector_inputs(request, &client, &style_json, &tile_source, bbox, width).await?;
1146            inputs.push((style_val, sources, assets));
1147        }
1148        let mut fonts = if options.text == SvgText::Editable {
1149            let mut stacks: Vec<String> = inputs.iter().flat_map(|i| i.2.glyphs.stacks.keys().cloned()).collect();
1150            stacks.sort();
1151            stacks.dedup();
1152            Some(vector::fonts::resolve(&client, &stacks, options).await)
1153        } else {
1154            None
1155        };
1156        let mut defs = String::new();
1157        let mut body = String::new();
1158        for (k, (style_val, sources, assets)) in inputs.iter().enumerate() {
1159            let parts = vector::render::render_svg(style_val, sources, assets, bbox, width, height, &format!("s{k}_"), options.text, fonts.as_ref())
1160                .map_err(BasemapError::RenderError)?;
1161            if !parts.skipped_layer_types.is_empty() {
1162                log::warn!("SVG output does not draw these layer types yet: {}", parts.skipped_layer_types.join(", "));
1163            }
1164            defs.push_str(&parts.defs);
1165            let _ = write!(body, r#"<g id="style{k}">{}</g>"#, parts.body);
1166        }
1167        if let Some(f) = fonts.as_mut() {
1168            f.finish(&client, options).await;
1169            let css = f.font_face_css(options.embed_fonts);
1170            if !css.is_empty() {
1171                defs = format!("<style>{css}</style>{defs}");
1172            }
1173        }
1174        Ok(format!(
1175            concat!(
1176                r#"<svg xmlns="http://www.w3.org/2000/svg" xmlns:xlink="http://www.w3.org/1999/xlink" "#,
1177                r#"width="{w}" height="{h}" viewBox="0 0 {w} {h}">"#,
1178                r#"<defs>{defs}</defs>{body}</svg>"#
1179            ),
1180            w = width,
1181            h = height,
1182            defs = defs,
1183            body = body
1184        ))
1185    })
1186}
