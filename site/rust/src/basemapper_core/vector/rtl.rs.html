1//! Right-to-left text, as MapLibre GL JS draws it with the
2//! @mapbox/mapbox-gl-rtl-text plugin (0.3.0, ICU compiled to WebAssembly):
3//!
4//! - `apply_arabic_shaping`: ICU `u_shapeArabic` with
5//!   `U_SHAPE_LETTERS_SHAPE | U_SHAPE_TEXT_DIRECTION_LOGICAL`. Arabic letters
6//!   become their isolated / initial / medial / final presentation forms
7//!   (U+FB50–U+FDFF, U+FE70–U+FEFC) from their neighbours' joining types;
8//!   lam + alef becomes one ligature (the text shrinks); tashkeel are
9//!   transparent and take their own isolated or medial forms. Applied to
10//!   every label after `text-transform`.
11//! - `process_bidi`: ICU `ubidi` with a default-LTR paragraph level. Lines are
12//!   broken on the logical text first; each line is then reordered for
13//!   display (UAX #9 with bracket pairs, rule L1 for trailing whitespace),
14//!   right-to-left runs reversed with mirrored brackets, and bidi controls
15//!   removed (`UBIDI_DO_MIRRORING | UBIDI_REMOVE_BIDI_CONTROLS`).
16//!
17//! The shaping tables and the joining walk are ported from ICU's ushape.cpp
18//! (© Unicode, Inc. and others, Unicode License); the mirroring pairs are
19//! BidiMirroring.txt (Unicode 15).

236fn get_link(ch: u16) -> u16 {
237    match ch {
238        0x0622..=0x06D3 => ARA_LINK[(ch - 0x0622) as usize],

242        0xFE70..=0xFEFC => PRES_B_LINK[(ch - 0xFE70) as usize] as u16,
243        _ => 0,
244    }

265/// countSpaces: spaces at each end (none at the right when all are spaces).
266fn count_spaces(s: &[u16]) -> (usize, usize) {
267    let l = s.iter().take_while(|&&c| c == SPACE_CHAR).count();
268    if l == s.len() {
269        return (l, 0);
270    }
271    (l, s.iter().rev().take_while(|&&c| c == SPACE_CHAR).count())
272}
273
274/// invertBuffer: reverses the text between the leading and trailing spaces.
275fn invert(s: &mut [u16]) {
276    let (l, r) = count_spaces(s);
277    let end = s.len() - r;
278    if l < end {
279        s[l..end].reverse();
280    }
281}

283/// calculateSize with U_SHAPE_LAMALEF_RESIZE, logical order: the length
284/// ICU reserves (one less per lam followed by alef, or presentation-form
285/// tashkeel).
286fn calculate_size(s: &[u16]) -> usize {
287    let mut n = s.len();
288    for i in 0..s.len() {
289        if (s[i] == LAM_CHAR && i + 1 < s.len() && is_alef(s[i + 1])) || (0xFE70..=0xFE7F).contains(&s[i]) {
290            n -= 1;
291        }
292    }
293    n
294}

296/// shapeUnicode (tashkeel flag 1) over a buffer in reversed order, then the
297/// lam-alef placeholders removed (handleGeneratedSpaces, resize).
298fn shape_unicode(dest: &mut Vec<u16>) {
299    for c in dest.iter_mut() {
300        if (0xFB50..=0xFBFF).contains(c) {
301            let m = CONVERT_FB_TO_06[(*c - 0xFB50) as usize];
302            if m != 0 {
303                *c = m;
304            }
305        } else if (0xFE70..=0xFEFC).contains(c) {
306            *c = CONVERT_FE_TO_06[(*c - 0xFE70) as usize];
307        }
308    }
309    let n = dest.len() as isize;
310    let (iend, step) = (-1isize, -1isize);
311    let mut i = n - 1;
312    let at = |d: &Vec<u16>, k: isize| d[k as usize];
313    let mut curr_link = get_link(at(dest, i));
314    let mut last_pos = i;
315    let (mut nx, mut nw): (isize, isize);
316    nx = -2;
317    let (mut prev_link, mut last_link, mut next_link) = (0u16, 0u16, 0u16);
318    let mut lamalef_found = false;
319    while i != iend {
320        if (curr_link & 0xFF00) > 0 || (get_link(at(dest, i)) & IRRELEVANT) != 0 {
321            nw = i + step;
322            while nx < 0 {
323                if nw == iend {
324                    next_link = 0;
325                    nx = 3000;
326                } else {
327                    next_link = get_link(at(dest, nw));
328                    if (next_link & IRRELEVANT) == 0 {
329                        nx = nw;
330                    } else {
331                        nw += step;
332                    }
333                }
334            }
335            if (curr_link & ALEFTYPE) > 0 && (last_link & LAMTYPE) > 0 {
336                lamalef_found = true;
337                let w = change_lam_alef(at(dest, i));
338                if w != 0 {
339                    dest[i as usize] = LAMALEF_SPACE_SUB;
340                    dest[last_pos as usize] = w;
341                    i = last_pos;
342                }
343                last_link = prev_link;
344                curr_link = get_link(w);
345            }
346            let mut shape = SHAPE_TABLE[(next_link & (LINKR + LINKL)) as usize][(last_link & (LINKR + LINKL)) as usize]
347                [(curr_link & (LINKR + LINKL)) as usize];
348            let c = at(dest, i);
349            if (curr_link & (LINKR + LINKL)) == 1 {
350                shape &= 1;
351            } else if is_tashkeel(c) {
352                if (last_link & LINKL) != 0 && (next_link & LINKR) != 0 && c != 0x064C && c != 0x064D {
353                    shape = 1;
354                    if (next_link & ALEFTYPE) == ALEFTYPE && (last_link & LAMTYPE) == LAMTYPE {
355                        shape = 0;
356                    }
357                } else {
358                    shape = 0;
359                }
360            }
361            if (c ^ 0x0600) < 0x100 {
362                if is_tashkeel(c) {
363                    dest[i as usize] = 0xFE70 + IRRELEVANT_POS[(c - 0x064B) as usize] + shape;
364                } else if (curr_link & APRESENT) > 0 {
365                    dest[i as usize] = 0xFB50 + (curr_link >> 8) + shape;
366                } else if (curr_link >> 8) > 0 && (curr_link & IRRELEVANT) == 0 {
367                    dest[i as usize] = 0xFE70 + (curr_link >> 8) + shape;
368                }
369            }
370        }
371        if (curr_link & IRRELEVANT) == 0 {
372            prev_link = last_link;
373            last_link = curr_link;
374            last_pos = i;
375        }
376        i += step;
377        if i == nx {
378            curr_link = next_link;
379            nx = -2;
380        } else if i != iend {
381            curr_link = get_link(at(dest, i));
382        }
383    }
384    if lamalef_found {
385        dest.retain(|&c| c != LAMALEF_SPACE_SUB);
386        // u_strlen: the text ends at a NUL.
387        if let Some(z) = dest.iter().position(|&c| c == 0) {
388            dest.truncate(z);
389        }
390    }
391}

393/// The plugin's applyArabicShaping (ushape_arabic).
394pub fn apply_arabic_shaping(input: &str) -> String {
395    let src: Vec<u16> = input.encode_utf16().collect();
396    // Only the 06xx letters change; presentation forms are re-shaped.
397    if !src.iter().any(|&c| (0x0600..=0x06FF).contains(&c) || (0xFB50..=0xFEFC).contains(&c)) {
398        return input.to_owned();
399    }
400    let reserved = calculate_size(&src);
401    let mut buf = src;
402    invert(&mut buf);
403    shape_unicode(&mut buf);
404    invert(&mut buf);
405    // The plugin reserves `reserved + 1` units and terminates at
406    // `reserved`; a longer result is an overflow error (input unchanged).
407    if buf.len() > reserved + 1 {
408        return input.to_owned();
409    }
410    buf.truncate(reserved);
411    String::from_utf16_lossy(&buf)
412}

414fn mirror(c: char) -> char {
415    match MIRRORS.binary_search_by_key(&(c as u32), |p| p.0) {
416        Ok(k) => char::from_u32(MIRRORS[k].1).unwrap_or(c),
417        Err(_) => c,
418    }
419}

432/// The plugin's processBidirectionalText: `chars` in logical order with
433/// line breaks (char indices, the last one the text length) chosen on it;
434/// returns each line in visual order. Paragraph ends (after a paragraph
435/// separator) are merged into the breaks as the plugin does.
436pub fn process_bidi(chars: &[char], breaks: &[usize]) -> Vec<Vec<char>> {
437    if chars.is_empty() {
438        return vec![Vec::new()];
439    }
440    let text: String = chars.iter().collect();
441    // Byte offset of each char index (and of the end).
442    let mut byte_at: Vec<usize> = Vec::with_capacity(chars.len() + 1);
443    let mut b = 0;
444    for c in chars {
445        byte_at.push(b);
446        b += c.len_utf8();
447    }
448    byte_at.push(b);
449    let char_at = |byte: usize| byte_at.partition_point(|&x| x < byte);
450    let info = BidiInfo::new(&text, None);
451    // mergeParagraphLineBreakPoints.
452    let mut merged: Vec<usize> = Vec::new();
453    for p in &info.paragraphs {
454        let end = char_at(p.range.end);
455        for &lb in breaks {
456            if lb < end && merged.last().is_none_or(|&m| lb > m) {
457                merged.push(lb);
458            }
459        }
460        merged.push(end);
461    }
462    for &lb in breaks {
463        if merged.last().is_none_or(|&m| lb > m) {
464            merged.push(lb);
465        }
466    }
467    let mut lines = Vec::with_capacity(merged.len());
468    let mut start = 0;
469    for &end in &merged {
470        let end = end.min(chars.len());
471        let mut line: Vec<char> = Vec::with_capacity(end.saturating_sub(start));
472        if start < end {
473            let (b0, b1) = (byte_at[start], byte_at[end]);
474            let para = info.paragraphs.iter().find(|p| p.range.start <= b0 && b0 < p.range.end).unwrap_or(&info.paragraphs[0]);
475            let b1 = b1.min(para.range.end);
476            let (levels, runs) = info.visual_runs(para, b0..b1);
477            for run in runs {
478                let s = &text[run.clone()];
479                if levels[run.start].is_rtl() {
480                    line.extend(s.chars().rev().map(mirror));
481                } else {
482                    line.extend(s.chars());
483                }
484            }
485            line.retain(|&c| !is_bidi_control(c));
486        }
487        lines.push(line);
488        start = end.max(start);
489    }
490    lines
491}
