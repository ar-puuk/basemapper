1//! Line-placed symbols: a port of the line parts of MapLibre GL JS's symbol
2//! pipeline.
3//!
4//! - `merge_lines`       symbol/merge_lines.ts
5//! - `clip_line`         symbol/clip_line.ts (clipLine)
6//! - `get_anchors`       symbol/get_anchors.ts (+ check_max_angle.ts)
7//! - `place_glyph_along_line`, `place_glyphs_along_line`
8//!                       symbol/projection.ts (label plane = screen pixels;
9//!                       no pitch, so projection is affine)
10//! - `collision_path`    the projected path used by placeCollisionCircles
11//!
12//! All geometry here is in MapLibre's 8192-unit tile space unless noted.

14use std::collections::HashMap;
15
16pub type P = (f64, f64);
17
18fn dist(a: P, b: P) -> f64 {
19    ((a.0 - b.0).powi(2) + (a.1 - b.1).powi(2)).sqrt()
20}
21
22/// point-geometry `a.angleTo(b)` = atan2(a.y - b.y, a.x - b.x).
23fn angle_to(a: P, b: P) -> f64 {
24    (a.1 - b.1).atan2(a.0 - b.0)
25}
26
27fn js_round(x: f64) -> f64 {
28    (x + 0.5).floor()
29}

33/// A line-placed feature during layout: its (possibly merged) geometry.

36    pub text: Option<String>,
37    pub geometry: Vec<Vec<P>>,
38}

40/// mergeLines: join features whose first lines share an endpoint and text.
41pub fn merge_lines(features: Vec<LineFeature>) -> Vec<LineFeature> {
42    let mut left: HashMap<String, usize> = HashMap::new();
43    let mut right: HashMap<String, usize> = HashMap::new();
44    let mut merged: Vec<Option<LineFeature>> = Vec::new();
45    let key = |text: &str, geom: &[Vec<P>], on_right: bool| {
46        let p = if on_right { *geom[0].last().unwrap() } else { geom[0][0] };
47        format!("{text}:{}:{}", p.0, p.1)
48    };
49    for f in features {
50        let text = match &f.text {
51            Some(t) if !t.is_empty() && !f.geometry.is_empty() && !f.geometry[0].is_empty() => t.clone(),
52            _ => {
53                merged.push(Some(f));
54                continue;
55            }
56        };
57        let lk = key(&text, &f.geometry, false);
58        let rk = key(&text, &f.geometry, true);
59        let in_right = right.get(&lk).copied();
60        let in_left = left.get(&rk).copied();
61        match (in_right, in_left) {
62            (Some(ri), Some(li)) if ri != li => {
63                // mergeFromLeft(lk, rk, geom)
64                left.remove(&rk);
65                left.insert(lk.clone(), li);
66                {
67                    let m = merged[li].as_mut().unwrap();
68                    m.geometry[0].remove(0);
69                    let mut g = f.geometry[0].clone();
70                    g.extend(m.geometry[0].iter().copied());
71                    m.geometry[0] = g;
72                }
73                // mergeFromRight(lk, rk, merged[li].geometry)
74                let geom_j = merged[li].as_ref().unwrap().geometry[0].clone();
75                right.remove(&lk);
76                right.insert(rk.clone(), ri);
77                {
78                    let m = merged[ri].as_mut().unwrap();
79                    m.geometry[0].pop();
80                    m.geometry[0].extend(geom_j);
81                }
82                left.remove(&lk);
83                right.remove(&rk);
84                let new_rk = key(&text, &merged[ri].as_ref().unwrap().geometry, true);
85                right.insert(new_rk, ri);
86                merged[li] = None;
87            }
88            (Some(ri), _) => {
89                right.remove(&lk);
90                right.insert(rk, ri);
91                let m = merged[ri].as_mut().unwrap();
92                m.geometry[0].pop();
93                m.geometry[0].extend(f.geometry[0].iter().copied());
94            }
95            (None, Some(li)) => {
96                left.remove(&rk);
97                left.insert(lk, li);
98                let m = merged[li].as_mut().unwrap();
99                m.geometry[0].remove(0);
100                let mut g = f.geometry[0].clone();
101                g.extend(m.geometry[0].iter().copied());
102                m.geometry[0] = g;
103            }
104            (None, None) => {
105                merged.push(Some(f));
106                let i = merged.len() - 1;
107                left.insert(lk, i);
108                right.insert(rk, i);
109            }
110        }
111    }
112    merged.into_iter().flatten().collect()
113}

117pub fn clip_line(lines: &[Vec<P>], x1: f64, y1: f64, x2: f64, y2: f64) -> Vec<Vec<P>> {
118    let mut out: Vec<Vec<P>> = Vec::new();
119    for line in lines {
120        let mut current: Option<usize> = None;
121        for i in 0..line.len().saturating_sub(1) {
122            let (mut p0, mut p1) = (line[i], line[i + 1]);
123            let r = |p: P| (js_round(p.0), js_round(p.1));
124            if p0.0 < x1 && p1.0 < x1 {
125                continue;
126            } else if p0.0 < x1 {
127                p0 = r((x1, p0.1 + (p1.1 - p0.1) * ((x1 - p0.0) / (p1.0 - p0.0))));
128            } else if p1.0 < x1 {
129                p1 = r((x1, p0.1 + (p1.1 - p0.1) * ((x1 - p0.0) / (p1.0 - p0.0))));
130            }
131            if p0.1 < y1 && p1.1 < y1 {
132                continue;
133            } else if p0.1 < y1 {
134                p0 = r((p0.0 + (p1.0 - p0.0) * ((y1 - p0.1) / (p1.1 - p0.1)), y1));
135            } else if p1.1 < y1 {
136                p1 = r((p0.0 + (p1.0 - p0.0) * ((y1 - p0.1) / (p1.1 - p0.1)), y1));
137            }
138            if p0.0 >= x2 && p1.0 >= x2 {
139                continue;
140            } else if p0.0 >= x2 {
141                p0 = r((x2, p0.1 + (p1.1 - p0.1) * ((x2 - p0.0) / (p1.0 - p0.0))));
142            } else if p1.0 >= x2 {
143                p1 = r((x2, p0.1 + (p1.1 - p0.1) * ((x2 - p0.0) / (p1.0 - p0.0))));
144            }
145            if p0.1 >= y2 && p1.1 >= y2 {
146                continue;
147            } else if p0.1 >= y2 {
148                p0 = r((p0.0 + (p1.0 - p0.0) * ((y2 - p0.1) / (p1.1 - p0.1)), y2));
149            } else if p1.1 >= y2 {
150                p1 = r((p0.0 + (p1.0 - p0.0) * ((y2 - p0.1) / (p1.1 - p0.1)), y2));
151            }
152            let start_new = match current {
153                None => true,
154                Some(c) => *out[c].last().unwrap() != p0,
155            };
156            if start_new {
157                out.push(vec![p0]);
158                current = Some(out.len() - 1);
159            }
160            out[current.unwrap()].push(p1);
161        }
162    }
163    out
164}

168#[derive(Clone, Copy, Debug)]
169pub struct Anchor {
170    pub x: f64,
171    pub y: f64,
172    pub segment: usize,
173}

175fn line_length(line: &[P]) -> f64 {
176    line.windows(2).map(|w| dist(w[0], w[1])).sum()
177}
178
179pub fn check_max_angle(line: &[P], anchor: &Anchor, label_length: f64, window_size: f64, max_angle: f64) -> bool {
180    if label_length == 0.0 {
181        return true;
182    }
183    let mut p = (anchor.x, anchor.y);
184    let mut index = anchor.segment as i64 + 1;
185    let mut anchor_distance = 0.0;
186    while anchor_distance > -label_length / 2.0 {
187        index -= 1;
188        if index < 0 {
189            return false;
190        }
191        anchor_distance -= dist(line[index as usize], p);
192        p = line[index as usize];
193    }
194    anchor_distance += dist(line[index as usize], line[index as usize + 1]);
195    index += 1;
196    let mut recent: std::collections::VecDeque<(f64, f64)> = std::collections::VecDeque::new();
197    let mut recent_delta = 0.0;
198    while anchor_distance < label_length / 2.0 {
199        let i = index as usize;
200        let prev = line[i - 1];
201        let current = line[i];
202        let Some(&next) = line.get(i + 1) else { return false };
203        let mut delta = angle_to(prev, current) - angle_to(current, next);
204        let tau = std::f64::consts::PI * 2.0;
205        // JS % keeps the sign of the dividend.
206        let m = (delta + 3.0 * std::f64::consts::PI) % tau;
207        delta = (m - std::f64::consts::PI).abs();
208        recent.push_back((anchor_distance, delta));
209        recent_delta += delta;
210        while anchor_distance - recent.front().unwrap().0 > window_size {
211            recent_delta -= recent.pop_front().unwrap().1;
212        }
213        if recent_delta > max_angle {
214            return false;
215        }
216        index += 1;
217        anchor_distance += dist(current, next);
218    }
219    true
220}

222#[allow(clippy::too_many_arguments)]
223fn resample(
224    line: &[P],
225    offset: f64,
226    spacing: f64,
227    angle_window: f64,
228    max_angle: f64,
229    label_length: f64,
230    continued: bool,
231    place_at_middle: bool,
232    extent: f64,
233) -> Vec<Anchor> {
234    let half = label_length / 2.0;
235    let total = line_length(line);
236    let mut distance = 0.0;
237    let mut marked = offset - spacing;
238    let mut anchors = Vec::new();
239    for i in 0..line.len().saturating_sub(1) {
240        let (a, b) = (line[i], line[i + 1]);
241        let seg = dist(a, b);
242        while marked + spacing < distance + seg {
243            marked += spacing;
244            let t = (marked - distance) / seg;
245            let x = a.0 + (b.0 - a.0) * t;
246            let y = a.1 + (b.1 - a.1) * t;
247            if x >= 0.0 && x < extent && y >= 0.0 && y < extent && marked - half >= 0.0 && marked + half <= total {
248                let anchor = Anchor { x: js_round(x), y: js_round(y), segment: i };
249                if angle_window == 0.0 || check_max_angle(line, &anchor, label_length, angle_window, max_angle) {
250                    anchors.push(anchor);
251                }
252            }
253        }
254        distance += seg;
255    }
256    if !place_at_middle && anchors.is_empty() && !continued {
257        anchors = resample(line, distance / 2.0, spacing, angle_window, max_angle, label_length, continued, true, extent);
258    }
259    anchors
260}

262/// getAnchors. `label_len` is the shaped label length (max of text and icon)
263/// in glyph units; `has_text` enables the angle window.
264#[allow(clippy::too_many_arguments)]
265pub fn get_anchors(
266    line: &[P],
267    spacing: f64,
268    max_angle: f64,
269    has_text: bool,
270    label_len: f64,
271    glyph_size: f64,
272    box_scale: f64,
273    overscaling: f64,
274    extent: f64,
275) -> Vec<Anchor> {
276    let angle_window = if has_text { 3.0 / 5.0 * glyph_size * box_scale } else { 0.0 };
277    let label_length = label_len * box_scale;
278    let continued = line[0].0 == 0.0 || line[0].0 == extent || line[0].1 == 0.0 || line[0].1 == extent;
279    let mut spacing = spacing;
280    if spacing - label_length < spacing / 4.0 {
281        spacing = label_length + spacing / 4.0;
282    }
283    let fixed_extra = glyph_size * 2.0;
284    let offset = if !continued {
285        ((label_len / 2.0 + fixed_extra) * box_scale * overscaling) % spacing
286    } else {
287        (spacing / 2.0 * overscaling) % spacing
288    };
289    resample(line, offset, spacing, angle_window, max_angle, label_length, continued, false, extent)
290}

292/// getCenterAnchor (symbol-placement: line-center).
293pub fn get_center_anchor(line: &[P], max_angle: f64, has_text: bool, label_len: f64, glyph_size: f64, box_scale: f64) -> Option<Anchor> {
294    let angle_window = if has_text { 3.0 / 5.0 * glyph_size * box_scale } else { 0.0 };
295    let label_length = label_len * box_scale;
296    let center = line_length(line) / 2.0;
297    let mut prev = 0.0;
298    for i in 0..line.len().saturating_sub(1) {
299        let (a, b) = (line[i], line[i + 1]);
300        let seg = dist(a, b);
301        if prev + seg > center {
302            let t = (center - prev) / seg;
303            let anchor = Anchor { x: js_round(a.0 + (b.0 - a.0) * t), y: js_round(a.1 + (b.1 - a.1) * t), segment: i };
304            return if angle_window == 0.0 || check_max_angle(line, &anchor, label_length, angle_window, max_angle) {
305                Some(anchor)
306            } else {
307                None
308            };
309        }
310        prev += seg;
311    }
312    None
313}

325#[derive(Default)]

327    offsets: HashMap<usize, P>,
328}

330#[derive(Clone, Debug)]
331pub struct PlacedGlyph {
332    pub point: P,
333    pub angle: f64,
334    pub path: Vec<P>,
335}
336
337fn unit(p: P) -> P {
338    let m = (p.0 * p.0 + p.1 * p.1).sqrt();
339    (p.0 / m, p.1 / m)
340}
341
342fn perp(p: P) -> P {
343    (-p.1, p.0)
344}
345
346fn offset_normal(seg: P, offset: f64, direction: f64) -> P {
347    let n = perp(unit(seg));
348    (n.0 * offset * direction, n.1 * offset * direction)
349}
350
351fn line_intersection(a1: P, a2: P, b1: P, b2: P) -> Option<P> {
352    let (ady, adx) = (a2.1 - a1.1, a2.0 - a1.0);
353    let (bdy, bdx) = (b2.1 - b1.1, b2.0 - b1.0);
354    let den = bdy * adx - bdx * ady;
355    if den == 0.0 {
356        return None;
357    }
358    let (ody, odx) = (a1.1 - b1.1, a1.0 - b1.0);
359    let t = (bdx * ody - bdy * odx) / den;
360    Some((a1.0 + t * adx, a1.1 + t * ady))
361}
362
363#[allow(clippy::too_many_arguments)]
364fn find_offset_intersection(
365    pl: &ProjectedLine,
366    cache: &mut ProjectionCache,
367    index: usize,
368    prev_to_current_normal: P,
369    current: P,
370    offset_prev: P,
371    line_offset_y: f64,
372    direction: i64,
373) -> P {
374    if let Some(&p) = cache.offsets.get(&index) {
375        return p;
376    }
377    let offset_current = (current.0 + prev_to_current_normal.0, current.1 + prev_to_current_normal.1);
378    let next_index = index as i64 + direction;
379    if next_index < 0 || next_index >= pl.line.len() as i64 {
380        cache.offsets.insert(index, offset_current);
381        return offset_current;
382    }
383    let next = pl.line[next_index as usize];
384    let n = offset_normal((next.0 - current.0, next.1 - current.1), line_offset_y, direction as f64);
385    let b1 = (current.0 + n.0, current.1 + n.1);
386    let b2 = (next.0 + n.0, next.1 + n.1);
387    let p = line_intersection(offset_prev, offset_current, b1, b2).unwrap_or(offset_current);
388    cache.offsets.insert(index, p);
389    p
390}

392/// placeGlyphAlongLine.
393#[allow(clippy::too_many_arguments)]
394pub fn place_glyph_along_line(
395    pl: &ProjectedLine,
396    cache: &mut ProjectionCache,
397    offset_x: f64,
398    line_offset_x: f64,
399    line_offset_y: f64,
400    flip: bool,
401    rotate_to_line: bool,
402) -> Option<PlacedGlyph> {
403    let combined = if flip { offset_x - line_offset_x } else { offset_x + line_offset_x };
404    let mut direction: i64 = if combined > 0.0 { 1 } else { -1 };
405    let mut angle = 0.0;
406    if flip {
407        direction *= -1;
408        angle = std::f64::consts::PI;
409    }
410    if direction < 0 {
411        angle += std::f64::consts::PI;
412    }
413    let mut current_index: i64 = if direction > 0 { pl.segment as i64 } else { pl.segment as i64 + 1 };
414    let anchor = pl.anchor;
415    let mut current = anchor;
416    let mut previous;
417    let mut offset_intersection: Option<P> = None;
418    let mut offset_prev: Option<P> = None;
419    let mut distance_from_anchor = 0.0;
420    let mut current_segment_distance = 0.0;
421    let abs_offset = combined.abs();
422    let mut path: Vec<P> = Vec::new();
423    let mut current_segment: P = (0.0, 0.0);
424    previous = anchor;
425
426    while distance_from_anchor + current_segment_distance <= abs_offset {
427        current_index += direction;
428        if current_index < 0 || current_index >= pl.line.len() as i64 {
429            return None;
430        }
431        distance_from_anchor += current_segment_distance;
432        previous = current;
433        offset_prev = offset_intersection;
434        current = pl.line[current_index as usize];
435        if line_offset_y == 0.0 {
436            path.push(previous);
437            current_segment = (current.0 - previous.0, current.1 - previous.1);
438        } else {
439            let prev_to_current = (current.0 - previous.0, current.1 - previous.1);
440            let normal = if prev_to_current.0 == 0.0 && prev_to_current.1 == 0.0 {
441                let ni = current_index + direction;
442                if ni < 0 || ni >= pl.line.len() as i64 {
443                    return None;
444                }
445                let next = pl.line[ni as usize];
446                offset_normal((next.0 - current.0, next.1 - current.1), line_offset_y, direction as f64)
447            } else {
448                offset_normal(prev_to_current, line_offset_y, direction as f64)
449            };
450            let op = offset_prev.unwrap_or((previous.0 + normal.0, previous.1 + normal.1));
451            offset_prev = Some(op);
452            let oi = find_offset_intersection(pl, cache, current_index as usize, normal, current, op, line_offset_y, direction);
453            offset_intersection = Some(oi);
454            path.push(op);
455            current_segment = (oi.0 - op.0, oi.1 - op.1);
456        }
457        current_segment_distance = (current_segment.0.powi(2) + current_segment.1.powi(2)).sqrt();
458    }
459    let t = (abs_offset - distance_from_anchor) / current_segment_distance;
460    let base = offset_prev.unwrap_or(previous);
461    let p = (current_segment.0 * t + base.0, current_segment.1 * t + base.1);
462    let segment_angle = angle + (current.1 - previous.1).atan2(current.0 - previous.0);
463    path.push(p);
464    Some(PlacedGlyph { point: p, angle: if rotate_to_line { segment_angle } else { 0.0 }, path })
465}

467/// Result of placeGlyphsAlongLine for drawing.
468pub enum GlyphsPlacement {
469    Hidden,
470    Placed(Vec<PlacedGlyph>),
471    /// A horizontal label with a vertical counterpart whose line runs more
472    /// vertically than horizontally: draw the vertical version instead.
473    UseVertical,
474}

476/// The writing mode of a symbol placed along a line (placedSymbol.writingMode).
477#[derive(Clone, Copy, PartialEq, Eq, Debug)]
478pub enum WritingMode {
479    /// Horizontal text without a vertical counterpart.
480    HorizontalOnly,
481    /// Horizontal text with a vertical counterpart.
482    Horizontal,
483    Vertical,
484}
485
486/// requiresOrientationChange from the first and last placed points (screen
487/// px, y down; MapLibre compares clip-space points scaled by the aspect
488/// ratio, the same as comparing pixels).
489fn orientation_change(mode: WritingMode, first: P, last: P) -> Option<Inner> {
490    if mode == WritingMode::Horizontal && (last.1 - first.1).abs() > (last.0 - first.0).abs() {
491        return Some(Inner::UseVertical);
492    }
493    let flip = if mode == WritingMode::Vertical { first.1 > last.1 } else { first.0 > last.0 };
494    flip.then_some(Inner::NeedsFlipping)
495}

497/// updateLineLabels for one symbol: place all glyphs (glyph offsets are in
498/// glyph units; `font_scale` = size / 24), flipping for keep-upright.
499pub fn place_glyphs_for_render(
500    pl: &ProjectedLine,
501    glyph_offsets: &[f64],
502    font_scale: f64,
503    line_offset: (f64, f64),
504    keep_upright: bool,
505    rotate_to_line: bool,
506    mode: WritingMode,
507) -> GlyphsPlacement {
508    let mut cache = ProjectionCache::default();
509    match place_glyphs(pl, &mut cache, glyph_offsets, font_scale, line_offset, false, keep_upright, rotate_to_line, mode) {
510        Inner::NotEnoughRoom => GlyphsPlacement::Hidden,
511        Inner::UseVertical => GlyphsPlacement::UseVertical,
512        Inner::NeedsFlipping => {
513            match place_glyphs(pl, &mut cache, glyph_offsets, font_scale, line_offset, true, keep_upright, rotate_to_line, mode) {
514                Inner::Placed(g) => GlyphsPlacement::Placed(g),
515                _ => GlyphsPlacement::Hidden,
516            }
517        }
518        Inner::Placed(g) => GlyphsPlacement::Placed(g),
519    }
520}
521
522enum Inner {
523    NotEnoughRoom,
524    NeedsFlipping,
525    UseVertical,
526    Placed(Vec<PlacedGlyph>),
527}

529#[allow(clippy::too_many_arguments)]
530fn place_glyphs(
531    pl: &ProjectedLine,
532    cache: &mut ProjectionCache,
533    offsets: &[f64],
534    font_scale: f64,
535    line_offset: (f64, f64),
536    flip: bool,
537    keep_upright: bool,
538    rotate_to_line: bool,
539    mode: WritingMode,
540) -> Inner {
541    let (lox, loy) = (line_offset.0 * font_scale, line_offset.1 * font_scale);
542    if offsets.len() > 1 {
543        let Some(first) = place_glyph_along_line(pl, cache, font_scale * offsets[0], lox, loy, flip, rotate_to_line) else {
544            return Inner::NotEnoughRoom;
545        };
546        let Some(last) =
547            place_glyph_along_line(pl, cache, font_scale * offsets[offsets.len() - 1], lox, loy, flip, rotate_to_line)
548        else {
549            return Inner::NotEnoughRoom;
550        };
551        if keep_upright && !flip {
552            if let Some(change) = orientation_change(mode, first.point, last.point) {
553                return change;
554            }
555        }
556        let mut placed = vec![first];
557        for &o in &offsets[1..offsets.len() - 1] {
558            match place_glyph_along_line(pl, cache, font_scale * o, lox, loy, flip, rotate_to_line) {
559                Some(g) => placed.push(g),
560                None => return Inner::NotEnoughRoom,
561            }
562        }
563        placed.push(last);
564        Inner::Placed(placed)
565    } else {
566        if keep_upright && !flip {
567            let a = pl.anchor;
568            let b = pl.line[(pl.segment + 1).min(pl.line.len() - 1)];
569            if let Some(change) = orientation_change(mode, a, b) {
570                return change;
571            }
572        }
573        match place_glyph_along_line(pl, cache, font_scale * offsets.first().copied().unwrap_or(0.0), lox, loy, flip, rotate_to_line)
574        {
575            Some(g) => Inner::Placed(vec![g]),
576            None => Inner::NotEnoughRoom,
577        }
578    }
579}

581/// The projected path placeCollisionCircles builds from the first and last
582/// glyph (flip = false, rotateToLine = false), or None if they don't fit.
583pub fn collision_path(pl: &ProjectedLine, glyph_offsets: &[f64], font_scale: f64, line_offset: (f64, f64)) -> Option<Vec<P>> {
584    let mut cache = ProjectionCache::default();
585    let (lox, loy) = (line_offset.0 * font_scale, line_offset.1 * font_scale);
586    let first = place_glyph_along_line(pl, &mut cache, font_scale * glyph_offsets[0], lox, loy, false, false)?;
587    let last = place_glyph_along_line(pl, &mut cache, font_scale * glyph_offsets[glyph_offsets.len() - 1], lox, loy, false, false)?;
588    let mut path = Vec::new();
589    for i in (1..first.path.len()).rev() {
590        path.push(first.path[i]);
591    }
592    for p in last.path.iter().skip(1) {
593        path.push(*p);
594    }
595    Some(path)
596}

598/// PathInterpolator.
599pub struct PathInterpolator {
600    points: Vec<P>,
601    distances: Vec<f64>,
602    pub length: f64,
603    pub padded_length: f64,
604    padding: f64,
605}
606
607impl PathInterpolator {
608    pub fn new(points: Vec<P>, padding: f64) -> Self {
609        let mut distances = vec![0.0];
610        for i in 1..points.len() {
611            distances.push(distances[i - 1] + dist(points[i], points[i - 1]));
612        }
613        let length = *distances.last().unwrap();
614        let padding = padding.min(length * 0.5);
615        PathInterpolator { points, distances, length, padded_length: length - padding * 2.0, padding }
616    }
617
618    pub fn lerp(&self, t: f64) -> P {
619        if self.points.len() == 1 {
620            return self.points[0];
621        }
622        let t = t.clamp(0.0, 1.0);
623        let mut idx = 1;
624        let mut d = self.distances[idx];
625        let target = t * self.padded_length + self.padding;
626        while d < target && idx < self.distances.len() {
627            idx += 1;
628            d = self.distances.get(idx).copied().unwrap_or(f64::NAN);
629        }
630        let prev = idx - 1;
631        let dp = self.distances[prev];
632        let seg = d - dp;
633        let st = if seg > 0.0 { (target - dp) / seg } else { 0.0 };
634        let cur = self.points.get(idx).copied().unwrap_or(self.points[prev]);
635        (self.points[prev].0 * (1.0 - st) + cur.0 * st, self.points[prev].1 * (1.0 - st) + cur.1 * st)
636    }
637}
