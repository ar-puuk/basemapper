1//! Vertical text, as MapLibre GL JS lays it out (util/script_detection.ts,
2//! util/verticalize_punctuation.ts, shaping.ts isLineVertical).
3//!
4//! Characters with upright vertical orientation (CJK ideographs, kana,
5//! Hangul, ...) stay upright along a vertical line: a label whose text has
6//! any such character gets a vertical shaping too, used for point labels
7//! with `text-writing-mode: vertical` and for line labels whenever their line
8//! runs more vertically than horizontally on screen. Punctuation between such
9//! characters becomes its vertical presentation form.
10//!
11//! The tables are generated from MapLibre's own functions
12//! (unicode_properties.g.ts, `\p{sc=Arab}` and `\s` regular expressions,
13//! verticalizedCharacterMap) evaluated over every code point.
14
15fn in_ranges(table: &[(u32, u32)], cp: u32) -> bool {
16    table.binary_search_by(|&(a, b)| if cp < a { std::cmp::Ordering::Greater } else if cp > b { std::cmp::Ordering::Less } else { std::cmp::Ordering::Equal }).is_ok()
17}
18
19/// codePointHasUprightVerticalOrientation.
20pub fn upright(cp: u32) -> bool {
21    in_ranges(&UPRIGHT, cp)
22}
23
24/// charHasRotatedVerticalOrientation.
25fn rotated(cp: u32) -> bool {
26    !(upright(cp) || in_ranges(&NEUTRAL, cp))
27}
28
29/// codePointAllowsIdeographicBreaking.
30pub fn ideographic_breaking(cp: u32) -> bool {
31    in_ranges(&IDEOGRAPHIC_BREAKING, cp)
32}
33
34/// allowsVerticalWritingMode: any character is upright.
35pub fn allows_vertical_writing_mode(text: &str) -> bool {
36    text.chars().any(|c| upright(c as u32))
37}
38
39/// The vertical presentation form of a punctuation character.
40pub fn verticalized(c: char) -> Option<char> {
41    VERTICAL_PUNCTUATION
42        .binary_search_by_key(&(c as u32), |p| p.0)
43        .ok()
44        .and_then(|k| char::from_u32(VERTICAL_PUNCTUATION[k].1))
45}
46
47/// verticalizePunctuation: punctuation whose neighbours are not rotated
48/// (or are punctuation themselves) takes its vertical form.
49pub fn verticalize_punctuation(chars: &[char]) -> Vec<char> {
50    let ok = |c: Option<&char>| c.is_none_or(|&c| !rotated(c as u32) || verticalized(c).is_some());
51    (0..chars.len())
52        .map(|i| {
53            let prev = if i == 0 { None } else { chars.get(i - 1) };
54            match verticalized(chars[i]) {
55                Some(v) if ok(chars.get(i + 1)) && ok(prev) => v,
56                _ => chars[i],
57            }
58        })
59        .collect()
60}
61
62/// isLineVertical for a glyph of a vertical shaping: upright characters, or
63/// with vertical point placement anything but whitespace and Arabic.
64pub fn glyph_is_vertical(cp: u32, allow_vertical_placement: bool) -> bool {
65    if allow_vertical_placement {
66        !(in_ranges(&WHITESPACE, cp) || in_ranges(&ARABIC_SCRIPT, cp))
67    } else {
68        upright(cp)
69    }
70}

72#[cfg(test)]
73mod tests {
74    use super::*;
75
76    #[test]
77    fn orientation_and_punctuation() {
78        assert!(upright('東' as u32) && upright('の' as u32) && upright('한' as u32));
79        assert!(!upright('A' as u32) && rotated('A' as u32));
80        assert!(allows_vertical_writing_mode("Tokyo 東京") && !allows_vertical_writing_mode("Tokyo"));
81        assert!(ideographic_breaking('京' as u32) && !ideographic_breaking('a' as u32));
82        let v: String = verticalize_punctuation(&"東京(駅)".chars().collect::<Vec<_>>()).into_iter().collect();
83        assert_eq!(v, "東京︵駅︶");
84        // Next to a Latin letter, punctuation stays as it is.
85        let v: String = verticalize_punctuation(&"A(B)".chars().collect::<Vec<_>>()).into_iter().collect();
86        assert_eq!(v, "A(B)");
87        assert!(glyph_is_vertical('A' as u32, true) && !glyph_is_vertical('A' as u32, false));
88        assert!(!glyph_is_vertical(' ' as u32, true) && !glyph_is_vertical(0x0628, true));
89    }
90}
