1//! Named basemaps and OGC service styles.
2//!
3//! The catalogue behind `basemap_provider()` and `list_providers()` in R and
4//! Python, so all three languages return the same style for the same name.
5//! Names follow leaflet-providers (`CartoDB.Positron`, `Esri.WorldImagery`,
6//! `OpenStreetMap.HOT`); basemaps leaflet lacks (OpenFreeMap) use the same
7//! `Provider.Variant` form. An entry has a vector form (a MapLibre style),
8//! a raster form (tiles), or both; the vector form is the default.
9//!
10//! Tile URLs, TileJSON, PMTiles archives and ArcGIS services need no entry:
11//! they are accepted as `style` directly, and [`crate::tile_style()`] builds
12//! their style with overrides.
13

18/// The raster form of a basemap.
19#[derive(Debug, Clone, Copy)]
20pub struct Raster {
21    /// Tile URL in leaflet form: `{s}` a subdomain, `{r}` `@2x` on
22    /// high-density displays (MapLibre's `{ratio}`), `{key}` the API key.
23    pub tiles: &'static str,
24    /// One subdomain per character, filled into `{s}`.
25    pub subdomains: &'static str,
26    pub maxzoom: u8,
27    /// Whether the tiles have labels drawn in.
28    pub labelled: bool,
29    /// The same map without labels, when the provider has it.
30    pub nolabels: Option<&'static str>,
31    /// The labels alone on transparent tiles, when the provider has them.
32    pub labels_only: Option<&'static str>,
33}

44/// A catalogue entry.
45#[derive(Debug, Clone, Copy)]
46pub struct Provider {

56/// Raster tiles with labels drawn in and no variants.
57const fn raster(tiles: &'static str, subdomains: &'static str, maxzoom: u8) -> Option<Raster> {
58    Some(Raster { tiles, subdomains, maxzoom, labelled: true, nolabels: None, labels_only: None })

61/// Raster tiles with labels, a label-free version and the labels alone.
62const fn raster_variants(

325/// Family names people use for the same provider.
326fn normalize(name: &str) -> String {
327    let lower = name.trim().to_ascii_lowercase();
328    let (family, rest) = lower.split_once('.').unwrap_or((lower.as_str(), ""));
329    let family = match family {
330        "carto" => "cartodb",
331        "osm" => "openstreetmap",
332        f => f,
333    };
334    if rest.is_empty() {
335        family.to_owned()
336    } else {
337        format!("{family}.{rest}")
338    }
339}

341fn levenshtein(a: &str, b: &str) -> usize {
342    let b: Vec<char> = b.chars().collect();
343    let mut row: Vec<usize> = (0..=b.len()).collect();
344    for (i, ca) in a.chars().enumerate() {
345        let mut prev = row[0];
346        row[0] = i + 1;
347        for (j, cb) in b.iter().enumerate() {
348            let cur = row[j + 1];
349            row[j + 1] = (prev + usize::from(ca != *cb)).min(row[j] + 1).min(cur + 1);
350            prev = cur;
351        }
352    }
353    row[b.len()]
354}

356/// The catalogue entry for `name` (case-insensitive; `Carto.` and `OSM.`
357/// work for `CartoDB.` and `OpenStreetMap.`, and earlier names still work).
358pub fn find_provider(name: &str) -> Option<&'static Provider> {
359    let key = normalize(name);
360    let target = ALIASES
361        .iter()
362        .find(|(alias, _)| normalize(alias) == key)
363        .map_or(key, |(_, to)| normalize(to));
364    PROVIDERS.iter().find(|p| normalize(p.name) == target)
365}
366
367/// Names close to an unknown `name`, best first.
368fn suggestions(name: &str) -> Vec<&'static str> {
369    let key = normalize(name);
370    let limit = (key.len() / 4).max(2);
371    let mut scored: Vec<(usize, &'static str)> = PROVIDERS
372        .iter()
373        .map(|p| (levenshtein(&key, &normalize(p.name)), p.name))
374        .filter(|(d, _)| *d <= limit)
375        .collect();
376    scored.sort();
377    let mut out: Vec<&'static str> = scored.into_iter().take(3).map(|(_, n)| n).collect();
378    if out.is_empty() {
379        // Same family, e.g. "CartoDB.Light": list its basemaps.
380        let family = key.split('.').next().unwrap_or("");
381        out = PROVIDERS
382            .iter()
383            .filter(|p| normalize(p.name).split('.').next() == Some(family))
384            .map(|p| p.name)
385            .collect();
386    }
387    out
388}

390fn unknown(name: &str) -> BasemapError {
391    let s = suggestions(name);
392    let hint = match s.len() {
393        0 => String::new(),
394        1 => format!("did you mean {}? ", s[0]),
395        _ => format!("did you mean one of {}? ", s.join(", ")),
396    };
397    BasemapError::InvalidInput(format!("unknown basemap '{name}'; {hint}See list_providers() for every name"))
398}

400/// The default style of a catalogue name given as `style`, e.g.
401/// `"CartoDB.Positron"`. `None` for anything that may be a path or URL; an
402/// error for a near miss of a name (with suggestions).
403pub(crate) fn bare_name_style(input: &str) -> Result<Option<String>, BasemapError> {
404    let s = input.trim();
405    let file_like = [".json", ".pmtiles", ".mbtiles", ".geojson", ".png", ".jpg", ".pbf", ".mvt"]
406        .iter()
407        .any(|ext| s.to_ascii_lowercase().ends_with(ext));
408    if s.is_empty() || file_like || s.contains(['/', '\\', ':', '{', '?']) || std::path::Path::new(s).exists() {
409        return Ok(None);
410    }
411    match find_provider(s) {
412        Some(p) => basemap_provider(p.name, &ProviderOptions::default()).map(Some),
413        None if !suggestions(s).is_empty() => Err(unknown(s)),
414        None => Ok(None),
415    }
416}

418impl Provider {
419    /// The forms this basemap comes in, vector first.
420    pub fn kinds(&self) -> Vec<TileKind> {
421        let mut k = Vec::new();
422        if self.vector.is_some() {
423            k.push(TileKind::Vector);
424        }
425        if self.raster.is_some() {
426            k.push(TileKind::Raster);
427        }
428        k
429    }
430}

432/// The basemaps that come in `kind` (all with `None`), in catalogue order.
433pub fn list_providers(kind: Option<TileKind>) -> Vec<&'static Provider> {
434    PROVIDERS.iter().filter(|p| kind.is_none_or(|k| p.kinds().contains(&k))).collect()
435}

437/// Options for [`basemap_provider`].
438#[derive(Debug, Clone, Default)]
439pub struct ProviderOptions {
440    /// Vector style or raster tiles; default the vector form when there is one.
441    pub kind: Option<TileKind>,
442    /// API key; default from the provider's environment variable.
443    pub api_key: Option<String>,
444    /// Tile data to draw instead of the provider's own (a `.pmtiles` archive,
445    /// tile URL or TileJSON, by URL or path), keeping its style. It must use
446    /// the same schema (source layers and fields) as the provider's tiles.
447    pub tiles: Option<String>,
448    /// Keep every layer, drop the labels, or keep only the labels. Raster
449    /// tiles switch to the provider's label-free or label-only tiles.
450    pub labels: crate::Labels,
451    /// Feature filter for the vector form (see [`crate::features`]).
452    pub features: Option<Vec<String>>,
453}
454
455fn other_kind(kind: TileKind) -> TileKind {
456    match kind {
457        TileKind::Vector => TileKind::Raster,
458        TileKind::Raster => TileKind::Vector,
459    }
460}

462/// A named basemap's style: a style URL for the vector form, inline style
463/// JSON for the raster form (or for either with `tiles`). See [`PROVIDERS`].
464pub fn basemap_provider(name: &str, options: &ProviderOptions) -> Result<String, BasemapError> {
465    let p = find_provider(name).ok_or_else(|| unknown(name))?;
466    let kind = options.kind.unwrap_or(p.kinds()[0]);
467    if !p.kinds().contains(&kind) {
468        let other = other_kind(kind);
469        let form = |k: TileKind| if k == TileKind::Vector { "a vector style" } else { "raster tiles" };
470        return Err(BasemapError::InvalidInput(format!(
471            "{} comes only as {}; use type \"{}\", or see list_providers(\"{}\") for basemaps with {}",
472            p.name,
473            form(other),
474            other.name(),
475            kind.name(),
476            form(kind),
477        )));
478    }
479    let key = match p.key {
480        None => String::new(),
481        Some(k) => options
482            .api_key
483            .clone()
484            .filter(|s| !s.is_empty())
485            .or_else(|| std::env::var(k.env).ok().filter(|s| !s.is_empty()))
486            .ok_or_else(|| {
487                BasemapError::InvalidInput(format!(
488                    "{} needs an API key: pass api_key or set the {} environment variable (keys: {})",
489                    p.name, k.env, k.signup
490                ))
491            })?,
492    };
493    let style = match kind {
494        TileKind::Vector => p.vector.unwrap_or_default().replace("{key}", &key),
495        TileKind::Raster => raster_style_for(p, &key),
496    };
497    let features = options.features.as_deref().unwrap_or(&[]);
498    let select = options.labels != crate::Labels::All || !features.is_empty();
499    if options.tiles.is_none() && (!select || kind == TileKind::Raster) {
500        return crate::features::select_layers(&style, options.labels, features);
501    }
502    crate::RUNTIME.block_on(async {
503        let client = crate::http_client(std::time::Duration::from_secs(30))?;
504        let style = match &options.tiles {
505            Some(tiles) => replace_tiles(&style, kind, tiles, &client).await?,
506            None => crate::tile_style::resolve_style_input(&style, &client, None).await?,
507        };
508        crate::features::select_layers(&style, options.labels, features)
509    })
510}

512/// The raster form as a style: one raster source, `{s}` expanded into one
513/// tile URL per subdomain as MapLibre spreads requests over `tiles`.
514fn raster_style_for(p: &Provider, key: &str) -> String {
515    let r = p.raster.expect("raster form");
516    let expand = |t: &str| -> Vec<String> {
517        let template = t.replace("{r}", "{ratio}").replace("{key}", key);
518        if r.subdomains.is_empty() {
519            vec![template]
520        } else {
521            r.subdomains.chars().map(|s| template.replace("{s}", &s.to_string())).collect()
522        }
523    };
524    let tiles = expand(r.tiles);
525    // The label variants, for `labels` (see crate::features::select_layers).
526    let mut variants = json!({"provider": p.name, "has_labels": r.labelled});
527    if let Some(t) = r.nolabels {
528        variants["nolabels"] = json!(expand(t));
529    }
530    if let Some(t) = r.labels_only {
531        variants["labels"] = json!(expand(t));
532    }
533    json!({
534        "metadata": { "basemapper:raster": variants },
535        "version": 8,
536        "sources": { "raster": {
537            "type": "raster",
538            "tiles": tiles,
539            "tileSize": 256,
540            "maxzoom": r.maxzoom,
541            "attribution": p.attribution,
542        }},
543        "layers": [{ "id": "raster", "type": "raster", "source": "raster" }],
544    })
545    .to_string()
546}

548/// `style` with every source of `kind` drawing `tiles` instead.
549async fn replace_tiles(
550    style: &str,
551    kind: TileKind,
552    tiles: &str,
553    client: &reqwest::Client,
554) -> Result<String, BasemapError> {
555    let new = match crate::tile_style::vector_tile_server(tiles) {
556        Some(base) => json!({"type": "vector", "url": base}),
557        None => {
558            // The source tile_style() builds for this data, without probing.
559            let opts = TileStyleOptions { kind: Some(kind), source_layers: Some(Vec::new()), ..Default::default() };
560            let (built, _) = crate::tile_style::tile_style(tiles, &opts, client, None).await?;
561            let v: Value = serde_json::from_str(&built).map_err(|e| BasemapError::StyleParseError(e.to_string()))?;
562            match v.pointer("/sources/tiles") {
563                Some(s) => s.clone(),
564                None => {
565                    return Err(BasemapError::InvalidInput(format!(
566                        "tiles {tiles:?} is a style; pass tile data (a .pmtiles archive, tile URL or TileJSON)"
567                    )))
568                }
569            }
570        }
571    };
572    let mut v: Value = if style.trim_start().starts_with('{') {
573        serde_json::from_str(style).map_err(|e| BasemapError::StyleParseError(e.to_string()))?
574    } else {
575        let resolved = crate::tile_style::resolve_style_input(style, client, None).await?;
576        serde_json::from_str(&resolved).map_err(|e| BasemapError::StyleParseError(e.to_string()))?
577    };
578    if let Some(m) = v["metadata"].as_object_mut() {
579        m.remove("basemapper:raster"); // the provider's label variants no longer apply
580    }
581    let by_url = new.get("url").is_some();
582    let mut replaced = 0;
583    if let Some(sources) = v["sources"].as_object_mut() {
584        for src in sources.values_mut() {
585            let Some(o) = src.as_object_mut() else { continue };
586            if o.get("type").and_then(Value::as_str) != Some(kind.name()) {
587                continue;
588            }
589            for k in ["url", "tiles", "scheme", "bounds", "attribution"] {
590                o.remove(k);
591            }
592            if by_url {
593                // A TileJSON or archive carries its own zoom range.
594                o.remove("minzoom");
595                o.remove("maxzoom");
596            }
597            for (k, val) in new.as_object().into_iter().flatten() {
598                if k != "type" {
599                    o.insert(k.clone(), val.clone());
600                }
601            }
602            replaced += 1;
603        }
604    }
605    if replaced == 0 {
606        return Err(BasemapError::InvalidInput(format!("the style has no {} source to draw tiles from", kind.name())));
607    }
608    Ok(v.to_string())
609}

611/// A style with one raster source and layer.
612fn raster_style(source: &str, layer: &str, tile_url: &str, tile_size: u32) -> String {
613    json!({
614        "version": 8,
615        "sources": { source: { "type": "raster", "tiles": [tile_url], "tileSize": tile_size } },
616        "layers": [{ "id": layer, "type": "raster", "source": source }],
617    })
618    .to_string()
619}

621/// Percent-encode a query value (everything but unreserved characters).
622fn encode_query(s: &str) -> String {
623    s.bytes()
624        .map(|b| match b {
625            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~' => (b as char).to_string(),
626            _ => format!("%{b:02X}"),
627        })
628        .collect()
629}

631/// A WMS GetMap endpoint as 256 px raster tiles in EPSG:3857, through
632/// MapLibre's `{bbox-epsg-3857}` token (filled in per tile). WMS 1.1.x names
633/// the projection `SRS`, 1.3.0 `CRS`.
634pub fn wms_style(
635    url: &str,
636    layers: &str,
637    styles: &str,
638    version: &str,
639    format: &str,
640    transparent: bool,
641    extra_params: Option<&str>,
642) -> String {
643    let crs = if version.starts_with("1.0") || version.starts_with("1.1") { "SRS" } else { "CRS" };
644    let mut tile_url = format!(
645        "{url}?SERVICE=WMS&VERSION={version}&REQUEST=GetMap&LAYERS={}&STYLES={}&{crs}=EPSG:3857\
646         &BBOX={{bbox-epsg-3857}}&WIDTH=256&HEIGHT=256&FORMAT={}&TRANSPARENT={}",
647        encode_query(layers),
648        encode_query(styles),
649        encode_query(format),
650        if transparent { "TRUE" } else { "FALSE" },
651    );
652    if let Some(extra) = extra_params.filter(|e| !e.is_empty()) {
653        tile_url.push('&');
654        tile_url.push_str(extra);
655    }
656    raster_style("wms", "wms-layer", &tile_url, 256)
657}

659/// A WMTS GetTile (KVP) endpoint as 256 px raster tiles.
660pub fn wmts_style(
661    url: &str,
662    layer: &str,
663    tilematrixset: &str,
664    format: &str,
665    style: &str,
666    version: &str,
667    extra_params: Option<&str>,
668) -> String {
669    let mut tile_url = format!(
670        "{url}?SERVICE=WMTS&REQUEST=GetTile&VERSION={version}&LAYER={}&STYLE={}&FORMAT={}&TILEMATRIXSET={}\
671         &TILEMATRIX={{z}}&TILEROW={{y}}&TILECOL={{x}}",
672        encode_query(layer),
673        encode_query(style),
674        encode_query(format),
675        encode_query(tilematrixset),
676    );
677    if let Some(extra) = extra_params.filter(|e| !e.is_empty()) {
678        tile_url.push('&');
679        tile_url.push_str(extra);
680    }
681    raster_style("wmts", "wmts-layer", &tile_url, 256)
682}
